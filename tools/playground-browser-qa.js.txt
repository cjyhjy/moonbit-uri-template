// Run through tabbit-cli nodejs on a task-owned tab after serving the built UI on 4184.
// The timeout case intentionally delays worker loading; it is not a real CPU hang.
const errors = [], requests = [], checks = [];
page.on('pageerror', e => errors.push(String(e)));
page.on('request', r => requests.push(r.url()));
await page.setViewportSize({width:1280,height:900});
await page.goto('http://127.0.0.1:4184/', {waitUntil:'domcontentloaded'});
const status = page.getByRole('status');
await expect(status).toHaveText('展开成功');
await expect(page.locator('#output')).toHaveText('/users/team%2Fa?tags=moon%20bit&tags=%E4%B8%AD%E6%96%87');
await page.locator('#profiles .location').first().click();
assert.equal(await page.getByLabel('URI 模板',{exact:true}).evaluate(el=>el.value.slice(el.selectionStart,el.selectionEnd)), 'id');
checks.push('desktop expansion and variable-reference selection');
await page.getByRole('button',{name:'遗漏预检',exact:true}).click();
await expect(status).toHaveText('需要修复');
await expect(page.locator('#error-message')).toContainText('MissingBinding');
await expect(page.locator('#copy')).toBeDisabled();
await page.getByRole('button',{name:'预算边界',exact:true}).click();
await expect(page.locator('#error-message')).toContainText('OutputLimit');
await page.getByLabel('最大输出字节',{exact:true}).fill('19');
await page.locator('#run').click();
await expect(status).toHaveText('展开成功');
await expect(page.locator('#bytes')).toHaveText('19 bytes');
checks.push('missing contract and exact UTF-8 output boundary');
await page.getByRole('button',{name:'错误定位',exact:true}).click();
await expect(page.locator('#error-message')).toContainText('UnclosedExpression');
await page.getByRole('button',{name:'定位到输入',exact:true}).click();
assert.equal(await page.getByLabel('URI 模板',{exact:true}).evaluate(el=>el.value.slice(el.selectionStart,el.selectionEnd)), '{id');
await page.getByRole('button',{name:'API 查询',exact:true}).click();
await expect(status).toHaveText('展开成功');
await page.getByLabel('变量 JSON',{exact:true}).fill('{"id":{"k":[]}}');
await expect(page.locator('#copy')).toBeDisabled();
await page.locator('#run').click();
await expect(page.locator('#error-message')).toContainText('InvalidValueType');
await page.getByRole('button',{name:'定位到输入',exact:true}).click();
assert.equal(await page.getByLabel('变量 JSON',{exact:true}).evaluate(el=>el.value.slice(el.selectionStart,el.selectionEnd)), '[]');
await page.getByLabel('变量 JSON',{exact:true}).fill('{"id":"😀","i\\u0064":"b"}');
await page.locator('#run').click();
await expect(page.locator('#error-message')).toContainText('DuplicateJsonKey');
checks.push('Unicode template and JSON selection; duplicate escaped keys rejected');
await page.getByRole('button',{name:'升级示例',exact:true}).click();
await expect(status).toHaveText('升级检查未通过');
await expect(page.locator('#changes')).toContainText('新增必填：page');
await expect(page.locator('#changes')).toContainText('截断长度');
await page.getByRole('button',{name:'定位到输入',exact:true}).click();
assert.equal(await page.getByLabel('候选 URI 模板',{exact:true}).evaluate(el=>el.value.slice(el.selectionStart,el.selectionEnd)), 'page');
await page.locator('#comparison').scrollIntoViewIfNeeded();
const desktop = await page.screenshot({fullPage:false});
await page.getByLabel('变量 JSON',{exact:true}).fill('{"id":"team/a","page":"2"}');
await expect(page.locator('#comparison')).toBeHidden();
await expect(page.locator('#copy')).toBeDisabled();
await page.locator('#run').click();
await expect(status).toHaveText('升级检查完成');
await expect(page.locator('#candidate-output')).toHaveText('/users/tea?page=2');
await page.getByLabel('候选最大输出字节',{exact:true}).fill('0');
await page.locator('#run').click();
await expect(status).toHaveText('升级检查未通过');
await expect(page.locator('#changes')).toContainText('maxOutput');
await expect(page.locator('#error-message')).toContainText('候选模板');
await page.getByLabel('候选 URI 模板',{exact:true}).fill('/users/{id}{?page}');
await page.getByLabel('候选必填变量',{exact:true}).fill('id');
await page.getByLabel('候选最大输出字节',{exact:true}).fill('8192');
await page.locator('#run').click();
await expect(status).toHaveText('升级检查完成');
await expect(page.locator('#comparison-status')).toHaveText('未发现契约变化');
checks.push('candidate missing -> repaired -> budget failure -> unchanged comparison; stale copy disabled');
await page.evaluate(()=>{ document.getElementById('run').click(); document.getElementById('cancel').click(); });
await expect(status).toHaveText('已取消计算');
await expect(page.locator('#copy')).toBeDisabled();
await expect(page.locator('#run')).toBeEnabled();
checks.push('actual worker cancellation');
let delayed = 0;
await page.route('**/worker.mjs', async route => {
  delayed++; await new Promise(resolve=>setTimeout(resolve,2500));
  await route.continue().catch(()=>{});
});
await page.locator('#run').click();
await expect(page.locator('#error-message')).toContainText('TimeLimit');
assert.equal(delayed,1);
await expect(page.locator('#copy')).toBeDisabled();
await page.unrouteAll({behavior:'wait'});
await page.locator('#run').click();
await expect(status).toHaveText('升级检查完成');
checks.push('controlled worker-load delay triggers timeout; retry succeeds');
await page.setViewportSize({width:390,height:844});
await page.getByRole('button',{name:'升级示例',exact:true}).click();
await expect(status).toHaveText('升级检查未通过');
await page.getByRole('heading',{name:'让每一个变量，都有着落。',exact:true}).scrollIntoViewIfNeeded();
assert(await page.evaluate(()=>document.documentElement.scrollWidth<=innerWidth));
const mobile = await page.screenshot({fullPage:false});
await page.getByRole('button',{name:'定位到输入',exact:true}).click();
assert.equal(await page.getByLabel('候选 URI 模板',{exact:true}).evaluate(el=>el.value.slice(el.selectionStart,el.selectionEnd)), 'page');
await page.locator('#comparison').scrollIntoViewIfNeeded();
const mobileResult = await page.screenshot({fullPage:false});
checks.push('390px viewport: no horizontal overflow; candidate selection and results usable');
assert.equal(errors.length,0);
const httpRequests = requests.filter(url=>/^https?:/.test(url));
assert(httpRequests.every(url=>url.startsWith('http://127.0.0.1:4184/')), JSON.stringify(httpRequests.filter(url=>!url.startsWith('http://127.0.0.1:4184/'))));
return {status:'passed',checks,browser:await page.evaluate(()=>navigator.userAgent),viewport:{width:390,height:844},pageErrors:errors,
  networkScope:'All observed HTTP(S) requests were local static assets; extension-scheme traffic is counted separately',requestCount:httpRequests.length,
  nonHttpSchemes:[...new Set(requests.filter(url=>!/^https?:/.test(url)).map(url=>url.split(':')[0]))],
  timeoutEvidence:'intentional 2500ms worker script delay against 2000ms deadline',screenshots:{desktop,mobile,mobileResult}};
